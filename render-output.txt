# Source: tazama/templates/infrastructure/postgres.yaml
apiVersion: v1
kind: ConfigMap
metadata:
  name: postgres-init
data:
  init.sql: |
    create database enrichment;
    
    -- Network Map Table - Official Tazama Schema
    -- Primary Key: (cfg, tenantId)
    -- Generated columns extracted from configuration jsonb
    create table network_map (
      configuration jsonb NOT NULL,
      cfg text GENERATED ALWAYS AS (configuration ->> 'cfg') STORED,
      tenantId text GENERATED ALWAYS AS (configuration ->> 'tenantId') STORED,
      active boolean GENERATED ALWAYS AS ((configuration ->> 'active')::boolean) STORED,
      creDtTm text GENERATED ALWAYS AS (configuration ->> 'creDtTm') STORED,
      updDtTm text GENERATED ALWAYS AS (configuration ->> 'updDtTm') STORED,
      PRIMARY KEY (cfg, tenantId)
    );
    
    create index idx_networkmap_active_tenant on network_map (tenantId) WHERE active = true;
    create index idx_networkmap_cre_dt_tm on network_map (creDtTm, tenantId);
    create index idx_networkmap_upd_dt_tm on network_map (updDtTm, tenantId);

    -- Typology Table - Official Tazama Schema
    -- Primary Key: (cfg, tenantId)
    create table typology (
      configuration jsonb NOT NULL,
      cfg text GENERATED ALWAYS AS (configuration ->> 'cfg') STORED,
      tenantId text GENERATED ALWAYS AS (configuration ->> 'tenantId') STORED,
      active boolean GENERATED ALWAYS AS ((configuration ->> 'active')::boolean) STORED,
      creDtTm text GENERATED ALWAYS AS (configuration ->> 'creDtTm') STORED,
      updDtTm text GENERATED ALWAYS AS (configuration ->> 'updDtTm') STORED,
      PRIMARY KEY (cfg, tenantId)
    );

    create index idx_typology_active_tenant on typology (tenantId) WHERE active = true;
    create index idx_typology_cre_dt_tm on typology (creDtTm, tenantId);
    create index idx_typology_upd_dt_tm on typology (updDtTm, tenantId);

    -- Rule Table - Official Tazama Schema
    -- Primary Key: (cfg, tenantId)
    create table rule (
      configuration jsonb NOT NULL,
      cfg text GENERATED ALWAYS AS (configuration ->> 'cfg') STORED,
      tenantId text GENERATED ALWAYS AS (configuration ->> 'tenantId') STORED,
      active boolean GENERATED ALWAYS AS ((configuration ->> 'active')::boolean) STORED,
      creDtTm text GENERATED ALWAYS AS (configuration ->> 'creDtTm') STORED,
      updDtTm text GENERATED ALWAYS AS (configuration ->> 'updDtTm') STORED,
      PRIMARY KEY (cfg, tenantId)
    );

    create index idx_rule_active_tenant on rule (tenantId) WHERE active = true;
    create index idx_rule_cre_dt_tm on rule (creDtTm, tenantId);
    create index idx_rule_upd_dt_tm on rule (updDtTm, tenantId);

    -- Evaluation Table - Official Tazama Schema
    -- Includes messageid extraction with coalesce for multiple message types
    create table evaluation (
      evaluation jsonb NOT NULL,
      messageid text GENERATED ALWAYS AS (
        coalesce(
          evaluation -> 'transaction' -> 'FIToFIPmtSts' -> 'GrpHdr' ->> 'MsgId',
          evaluation -> 'transaction' -> 'FIToFICstmrCdtTrf' -> 'GrpHdr' ->> 'MsgId',
          evaluation -> 'transaction' -> 'CstmrCdtTrfInitn' -> 'GrpHdr' ->> 'MsgId',
          evaluation -> 'transaction' -> 'CdtrPmtActvtnReq' -> 'GrpHdr' ->> 'MsgId'
        )
      ) STORED,
      tenantId text GENERATED ALWAYS AS (evaluation ->> 'tenantId') STORED,
      creDtTm text GENERATED ALWAYS AS (evaluation -> 'report' ->> 'timestamp') STORED
    );

    create index idx_evaluation_cre_dt_tm on evaluation (creDtTm, tenantId);

    -- Message Tables - Preserving original structure
    create table pacs002 (
      id uuid DEFAULT gen_random_uuid(),
      message jsonb NOT NULL,
      tenantId text NOT NULL,
      creDtTm text GENERATED ALWAYS AS (message -> 'GrpHdr' ->> 'CreDtTm') STORED,
      PRIMARY KEY (id, tenantId)
    );

    create index idx_pacs002_tenant on pacs002 (tenantId);
    create index idx_pacs002_cre_dt_tm on pacs002 (creDtTm);

    create table pacs008 (
      id uuid DEFAULT gen_random_uuid(),
      message jsonb NOT NULL,
      tenantId text NOT NULL,
      creDtTm text GENERATED ALWAYS AS (message -> 'GrpHdr' ->> 'CreDtTm') STORED,
      PRIMARY KEY (id, tenantId)
    );

    create index idx_pacs008_tenant on pacs008 (tenantId);
    create index idx_pacs008_cre_dt_tm on pacs008 (creDtTm);

    create table pain001 (
      id uuid DEFAULT gen_random_uuid(),
      message jsonb NOT NULL,
      tenantId text NOT NULL,
      creDtTm text GENERATED ALWAYS AS (message -> 'GrpHdr' ->> 'CreDtTm') STORED,
      PRIMARY KEY (id, tenantId)
    );

    create index idx_pain001_tenant on pain001 (tenantId);
    create index idx_pain001_cre_dt_tm on pain001 (creDtTm);

    create table pain013 (
      id uuid DEFAULT gen_random_uuid(),
      message jsonb NOT NULL,
      tenantId text NOT NULL,
      creDtTm text GENERATED ALWAYS AS (message -> 'GrpHdr' ->> 'CreDtTm') STORED,
      PRIMARY KEY (id, tenantId)
    );

    create index idx_pain013_tenant on pain013 (tenantId);
    create index idx_pain013_cre_dt_tm on pain013 (creDtTm);

    create table pacs013 (
      id uuid DEFAULT gen_random_uuid(),
      message jsonb NOT NULL,
      tenantId text NOT NULL,
      creDtTm text GENERATED ALWAYS AS (message -> 'GrpHdr' ->> 'CreDtTm') STORED,
      PRIMARY KEY (id, tenantId)
    );

    create index idx_pacs013_tenant on pacs013 (tenantId);
    create index idx_pacs013_cre_dt_tm on pacs013 (creDtTm);

    -- DEMS Quarantine Table - Official Tazama Schema
    -- For quarantine of failed DEMS deliveries
    create table public.dems_quarantine (
      id uuid NOT NULL,
      correlation_id varchar(255) NULL,
      tenant_id varchar(100) NOT NULL,
      endpoint_path text NULL,
      config_id varchar(255) NULL,
      version varchar(50) NULL,
      error text NULL,
      raw_payload jsonb NULL,
      status varchar(50) NULL,
      created_at timestamp NULL DEFAULT CURRENT_TIMESTAMP,
      PRIMARY KEY (id)
    );

    create index idx_dems_quarantine_tenant on dems_quarantine (tenant_id);
    create index idx_dems_quarantine_status on dems_quarantine (status);
    create index idx_dems_quarantine_created_at on dems_quarantine (created_at);

    -- Event History Tables - Preserving original structure
    create table event_history (
      id uuid DEFAULT gen_random_uuid(),
      event jsonb NOT NULL,
      tenantId text NOT NULL,
      creDtTm timestamp DEFAULT CURRENT_TIMESTAMP,
      PRIMARY KEY (id, tenantId)
    );

    create index idx_event_history_tenant on event_history (tenantId);
    create index idx_event_history_cre_dt_tm on event_history (creDtTm);

    create table alert_event_history (
      id uuid DEFAULT gen_random_uuid(),
      event jsonb NOT NULL,
      tenantId text NOT NULL,
      creDtTm timestamp DEFAULT CURRENT_TIMESTAMP,
      PRIMARY KEY (id, tenantId)
    );

    create index idx_alert_event_history_tenant on alert_event_history (tenantId);
    create index idx_alert_event_history_cre_dt_tm on alert_event_history (creDtTm);

  namespace: default
  labels:
    app: rule-063
    app.kubernetes.io/name: rule-063
    app.kubernetes.io/part-of: tazama
    app.kubernetes.io/component: rule
spec:
  replicas: 1
  selector:
    matchLabels:
      app: rule-063
  template:
    metadata:
      labels:
        app: rule-063
        app.kubernetes.io/name: rule-063
        app.kubernetes.io/part-of: tazama
        app.kubernetes.io/component: rule
      annotations:
        prometheus.io/scrape: "true"
        prometheus.io/port: "3000"
        prometheus.io/path: "/metrics"
    spec:
      initContainers:
        - name: wait-db
          image: "postgres:18-alpine"
          imagePullPolicy: IfNotPresent
          env:
            - name: PGPASSWORD
              valueFrom:
                secretKeyRef:
                  name: tazama-shared-secrets
                  key: CONFIGURATION_DATABASE_PASSWORD
          command:
            - sh
            - -c
            - |
              until psql -h postgres -U postgres -d raw_history -c "SELECT 'pain013'::regclass;" >/dev/null 2>&1; do
                echo "Waiting for the full Tazama database schema to finish initializing..."
                sleep 2
              done
              echo "Database schema ready"
        - name: wait-valkey
          image: "busybox:1.36"
          imagePullPolicy: IfNotPresent
          command:
            - sh
            - -c
            - |
              until nc -z valkey 6379; do
                echo "Waiting for Valkey..."
                sleep 2
              done
              echo "Valkey ready"
        - name: wait-nats
          image: "busybox:1.36"
          imagePullPolicy: IfNotPresent
          command:
            - sh
            - -c
            - |
              until nc -z nats 4222; do
                echo "Waiting for NATS..."
                sleep 2
              done
              echo "NATS ready"
      
      containers:
        - name: rule-063
          image: "tazamaorg/rule-063:rc"
          imagePullPolicy: IfNotPresent
          ports:
            - name: http
              containerPort: 3000
              protocol: TCP
              
          env:
            - name: STARTUP_TYPE
              value: "nats"
              
            - name: APM_SERVICE_NAME
              value: "rule-063"
            - name: CONFIGURATION_DATABASE
              value: "configuration"
            - name: RAW_HISTORY_DATABASE
              value: "raw_history"
            - name: EVENT_HISTORY_DATABASE
              value: "event_history"
            - name: CONFIGURATION_DATABASE_HOST
              value: "postgres"
            - name: CONFIGURATION_DATABASE_PORT
              value: "5432"
            - name: RAW_HISTORY_DATABASE_HOST
              value: "postgres"
            - name: RAW_HISTORY_DATABASE_PORT
              value: "5432"
            - name: EVENT_HISTORY_DATABASE_HOST
              value: "postgres"
            - name: EVENT_HISTORY_DATABASE_PORT
              value: "5432"
            - name: REDIS_SERVERS
              value: '[{"host":"valkey","port":6379}]'
            - name: REDIS_CLUSTER
              value: "false"
